    private void setupDragging() {
        canvas.setOnMousePressed(e -> {
            dragStartX = e.getSceneX();
            dragStartY = e.getSceneY();
        });

        canvas.setOnMouseDragged(e -> {
            double dx = e.getSceneX() - dragStartX;
            double dy = e.getSceneY() - dragStartY;

            spriteX += dx;
            spriteY += dy;

            dragStartX = e.getSceneX();
            dragStartY = e.getSceneY();

            drawSprite();
        });
    }

        }.start();
    }
